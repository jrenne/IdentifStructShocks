% Semantic, page-breakable callouts shared by the textbook PDF editions.
\usepackage{xcolor}
\usepackage{mdframed}
\definecolor{PracticeFill}{HTML}{EDF4F7}
\definecolor{PracticeRule}{HTML}{6C94A7}
\definecolor{ExerciseRule}{HTML}{9AAFB9}
\definecolor{RecapFill}{HTML}{F3F5F6}
\mdfdefinestyle{textbookbox}{
  skipabove=10pt,skipbelow=10pt,
  innerleftmargin=10pt,innerrightmargin=10pt,
  innertopmargin=8pt,innerbottommargin=8pt,
  splittopskip=10pt,splitbottomskip=8pt,
  linewidth=.6pt,roundcorner=0pt,
  nobreak=false
}
\newmdenv[style=textbookbox,backgroundcolor=PracticeFill,
  linecolor=PracticeRule,linewidth=2pt,
  topline=false,bottomline=false,rightline=false]{practicebox}
\newmdenv[style=textbookbox,backgroundcolor=white,
  linecolor=ExerciseRule,linewidth=.8pt,
  topline=true,bottomline=true,leftline=true,rightline=true]{exercisebox}
\newmdenv[style=textbookbox,backgroundcolor=RecapFill,
  linecolor=ExerciseRule,linewidth=0pt]{recapbox}

% Formal statements use a different treatment from practice and exercises.
% Proofs stay outside the boxes. Install the hooks after bookdown declares its
% theorem environments. H placement keeps figures inside their statement box.
\definecolor{DefinitionFill}{HTML}{F2F6F8}
\definecolor{ResultFill}{HTML}{F0F5F1}
\definecolor{ExampleFill}{HTML}{FAF6ED}
\definecolor{ExampleRule}{HTML}{B39A68}
\definecolor{ResultRule}{HTML}{70917C}
\mdfdefinestyle{definitionframe}{style=textbookbox,needspace=8\baselineskip,
  backgroundcolor=DefinitionFill,linecolor=ExerciseRule,linewidth=.8pt,
  topline=true,bottomline=true,leftline=true,rightline=true}
\mdfdefinestyle{resultframe}{style=textbookbox,needspace=8\baselineskip,
  backgroundcolor=ResultFill,linecolor=ResultRule,linewidth=.8pt,
  topline=true,bottomline=true,leftline=true,rightline=true}
% Examples may contain figures and span several pages.
\mdfdefinestyle{exampleframe}{style=textbookbox,needspace=12\baselineskip,
  backgroundcolor=ExampleFill,linecolor=ExampleRule,linewidth=.8pt,
  topline=true,bottomline=true,leftline=true,rightline=true}
\makeatletter
\AtBeginDocument{%
  \@ifundefined{example}{}{%
    \surroundwithmdframed[style=exampleframe]{example}%
    \AtBeginEnvironment{example}{\floatplacement{figure}{H}\floatplacement{table}{H}}%
  }%
  \@ifundefined{definition}{}{%
    \surroundwithmdframed[style=definitionframe]{definition}%
    \AtBeginEnvironment{definition}{\floatplacement{figure}{H}}%
  }%
  \@ifundefined{proposition}{}{%
    \surroundwithmdframed[style=resultframe,nobreak=true]{proposition}%
    \AtBeginEnvironment{proposition}{\floatplacement{figure}{H}}%
  }%
  \@ifundefined{theorem}{}{%
    \surroundwithmdframed[style=resultframe]{theorem}%
    \AtBeginEnvironment{theorem}{\floatplacement{figure}{H}}%
  }%
  \@ifundefined{hypothesis}{}{%
    \surroundwithmdframed[style=definitionframe]{hypothesis}%
    \AtBeginEnvironment{hypothesis}{\floatplacement{figure}{H}}%
  }%
}
\makeatother
